);

CREATE TABLE IF NOT EXISTS cinemas (

    cinema_name VARCHAR(50) NOT NULL,
    movie_id INTEGER NOT NULL
);

using Npgsql;
using System.Collections.ObjectModel;

namespace CinemaApp {

    // Data model for the CollectionView items
    public class MovieModel {
        public long Id { get; set; }
        public string Title { get; set; } = string.Empty;
        public string Genre { get; set; } = string.Empty;
        public int ReleaseYear { get; set; }
        public string Cinema { get; set; } = string.Empty;
    }

    public partial class MainPage : ContentPage {

        // Database connection settings
        private readonly NpgsqlConnectionStringBuilder connectionStringBuilder = new NpgsqlConnectionStringBuilder {
            Host = "localhost",
            Port = 5432,
            Username = "postgres",
            Password = "postgresql123",
            Database = "cinema_db",
        };

        // Collection used for dynamic data display
        private readonly ObservableCollection<MovieModel> movies = new ObservableCollection<MovieModel>();
        private bool hasLoadedData;

        public MainPage() {
            InitializeComponent();

            // Bind the collection to the CollectionView
            moviesCollectionView.ItemsSource = movies;

        }

        private async void MainPage_Loaded(object? sender, EventArgs e) {
            if (hasLoadedData) {
                return;
            }

            hasLoadedData = true;
            await LoadTable();
        }

        // Load data from the database into the CollectionView
        private async Task LoadTable() {
            try {
                using (var connection = new NpgsqlConnection(connectionStringBuilder.ConnectionString)) {
                    await connection.OpenAsync();

                    string query = "SELECT movies.id, movies.title, movies.genre, movies.release_year, cinemas.cinema_name FROM movies INNER JOIN cinemas ON movies.id = cinemas.movie_id";

                    using (var command = new NpgsqlCommand(query, connection))
                    using (var reader = await command.ExecuteReaderAsync()) {

                        // Collect the data in a temporary list
                        var movieItems = new List<MovieModel>();
                        while (await reader.ReadAsync()) {
                            movieItems.Add(new MovieModel {
                                Id = Convert.ToInt64(reader[0]),
                                Title = reader[1].ToString() ?? "",
                                Genre = reader[2].ToString() ?? "",
                                ReleaseYear = Convert.ToInt32(reader[3]),
                                Cinema = reader[4].ToString() ?? ""
                            });
                        }

                        // Update the UI on the main thread
                        MainThread.BeginInvokeOnMainThread(() => {
                            movies.Clear();
                            foreach (var movie in movieItems) {
                                movies.Add(movie);
                            }
                        });
                    }
                }
            } catch (Exception ex) {
                MainThread.BeginInvokeOnMainThread(async () => {
                    await DisplayAlertAsync("Error", "An error occurred while loading the data: " + ex.Message, "OK");
                });
            }
        }

        // Handle selection in the CollectionView
        private void MoviesCollectionView_SelectionChanged(object sender, SelectionChangedEventArgs e) {
            if (e.CurrentSelection.FirstOrDefault() is MovieModel selectedMovie) {
                selectedMovieIdLabel.Text = $"Selected movie ID: {selectedMovie.Id}";
                titleEntry.Text = selectedMovie.Title;
                genreEntry.Text = selectedMovie.Genre;
                releaseYearEntry.Text = selectedMovie.ReleaseYear.ToString();
                cinemaEntry.Text = selectedMovie.Cinema;
            }
        }

        // Add button
        private async void AddMovieButton_Clicked(object sender, EventArgs e) {
            // Validate the input fields
            if (string.IsNullOrWhiteSpace(titleEntry.Text) ||
                string.IsNullOrWhiteSpace(genreEntry.Text) ||
                string.IsNullOrWhiteSpace(releaseYearEntry.Text) ||
                string.IsNullOrWhiteSpace(cinemaEntry.Text)) {
                await DisplayAlertAsync("Warning", "All fields must be filled in.", "OK");
                return;
            }

            try {
                using (var connection = new NpgsqlConnection(connectionStringBuilder.ConnectionString)) {
                    await connection.OpenAsync();

                    // Insert the new movie
                    string insertMovieQuery = @"INSERT INTO movies (title, genre, release_year) 
                                               VALUES (@title, @genre, @release_year)
                                               RETURNING id;";

                    long newMovieId;
                    using (var command = new NpgsqlCommand(insertMovieQuery, connection)) {
                        command.Parameters.AddWithValue("@title", titleEntry.Text);
                        command.Parameters.AddWithValue("@genre", genreEntry.Text);
                        command.Parameters.AddWithValue("@release_year", int.Parse(releaseYearEntry.Text));

                        var result = await command.ExecuteScalarAsync();
                        newMovieId = Convert.ToInt64(result);
                    }

                    // Insert the associated cinema entry
                    string insertCinemaQuery = "INSERT INTO cinemas (movie_id, cinema_name) VALUES (@movie_id, @cinema_name)";
                    using (var command = new NpgsqlCommand(insertCinemaQuery, connection)) {
                        command.Parameters.AddWithValue("@movie_id", newMovieId);
                        command.Parameters.AddWithValue("@cinema_name", cinemaEntry.Text);
                        await command.ExecuteNonQueryAsync();
                    }

                    await DisplayAlertAsync("Success", "The movie was added successfully.", "OK");

                    ClearFields();
                    await LoadTable();
                }
            } catch (FormatException) {
                await DisplayAlertAsync("Error", "The release year must be a number.", "OK");
            } catch (Exception ex) {
                await DisplayAlertAsync("Error", "An error occurred while adding the movie: " + ex.Message, "OK");
            }

        }

        // Edit button
        private async void EditMovieButton_Clicked(object sender, EventArgs e) {
            if (moviesCollectionView.SelectedItem is not MovieModel selectedMovie) {
                await DisplayAlertAsync("Warning", "Please select a movie to edit from the list.", "OK");
                return;
            }

            if (string.IsNullOrWhiteSpace(titleEntry.Text) ||
                string.IsNullOrWhiteSpace(genreEntry.Text) ||
                string.IsNullOrWhiteSpace(releaseYearEntry.Text) ||
                string.IsNullOrWhiteSpace(cinemaEntry.Text)) {
                await DisplayAlertAsync("Warning", "All fields must be filled in to edit the movie.", "OK");
                return;
            }

            try {
                using (var connection = new NpgsqlConnection(connectionStringBuilder.ConnectionString)) {
                    await connection.OpenAsync();

                    // 1. Update the movie data
                    string updateMovieQuery = @"UPDATE movies 
                                               SET title = @title, genre = @genre, release_year = @release_year 
                                               WHERE id = @id";

                    using (var command = new NpgsqlCommand(updateMovieQuery, connection)) {
                        command.Parameters.AddWithValue("@title", titleEntry.Text);
                        command.Parameters.AddWithValue("@genre", genreEntry.Text);
                        command.Parameters.AddWithValue("@release_year", int.Parse(releaseYearEntry.Text));
                        command.Parameters.AddWithValue("@id", selectedMovie.Id);

                        await command.ExecuteNonQueryAsync();
                    }

                    // 2. Update the cinema entry
                    string updateCinemaQuery = "UPDATE cinemas SET cinema_name = @cinema_name WHERE movie_id = @id";
                    using (var command = new NpgsqlCommand(updateCinemaQuery, connection)) {
                        command.Parameters.AddWithValue("@cinema_name", cinemaEntry.Text);
                        command.Parameters.AddWithValue("@id", selectedMovie.Id);

                        await command.ExecuteNonQueryAsync();
                    }

                    await DisplayAlertAsync("Success", "The movie was updated successfully.", "OK");

                    ClearFields();
                    await LoadTable();
                }
            } catch (FormatException) {
                await DisplayAlertAsync("Error", "The release year must be a number.", "OK");
            } catch (Exception ex) {
                await DisplayAlertAsync("Error", "An error occurred while updating the movie: " + ex.Message, "OK");
            }
        }

        // Delete button
        private async void DeleteMovieButton_Clicked(object sender, EventArgs e) {
            if (moviesCollectionView.SelectedItem is not MovieModel selectedMovie) {
                await DisplayAlertAsync("Warning", "Please select a movie to delete from the list.", "OK");
                return;
            }

            bool isConfirmed = await DisplayAlertAsync("Confirmation", $"Are you sure you want to delete '{selectedMovie.Title}'?", "Yes", "No");
            if (!isConfirmed) return;

            try {
                using (var connection = new NpgsqlConnection(connectionStringBuilder.ConnectionString)) {
                    await connection.OpenAsync();

                    // Delete the related cinema row first unless ON DELETE CASCADE handles it
                    string deleteCinemaQuery = "DELETE FROM cinemas WHERE movie_id = @id";
                    using (var command = new NpgsqlCommand(deleteCinemaQuery, connection)) {
                        command.Parameters.AddWithValue("@id", selectedMovie.Id);
                        await command.ExecuteNonQueryAsync();
                    }

                    string deleteMovieQuery = "DELETE FROM movies WHERE id = @id";
                    using (var command = new NpgsqlCommand(deleteMovieQuery, connection)) {
                        command.Parameters.AddWithValue("@id", selectedMovie.Id);
                        await command.ExecuteNonQueryAsync();
                    }

                    await DisplayAlertAsync("Success", "The movie was deleted successfully.", "OK");

                    ClearFields();
                    await LoadTable();
                }
            } catch (Exception ex) {
                await DisplayAlertAsync("Error", "An error occurred while deleting the movie: " + ex.Message, "OK");
            }
        }

        private void ClearFields() {
            selectedMovieIdLabel.Text = "Selected movie ID: -";
            titleEntry.Text = string.Empty;
            genreEntry.Text = string.Empty;
            releaseYearEntry.Text = string.Empty;
            cinemaEntry.Text = string.Empty;
            moviesCollectionView.SelectedItem = null;
        }
    }
}
