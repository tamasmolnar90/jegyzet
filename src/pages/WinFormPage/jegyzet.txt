using System.ComponentModel;

namespace WinFormsApp1 {
    public partial class Form1 : Form {
        BindingList<string> listBoxItemsList = new BindingList<string>();

        public Form1() {
            InitializeComponent();
            listBox.DataSource = listBoxItemsList;
        }

        private void openToolStripMenuItem_Click(object sender, EventArgs e) {           
            using OpenFileDialog ofd = new OpenFileDialog();

            ofd.Filter = "All Files (*.*)|*.*";

            if (ofd.ShowDialog() == DialogResult.OK) {
                try {
                    listBoxItemsList.Clear();
                    string[] sorok = File.ReadAllLines(ofd.FileName);

                    foreach (var sor in sorok) {
                        listBoxItemsList.Add(sor);
                    }
                } catch (Exception ex) {
                    MessageBox.Show(
                        $"Hiba történt: {ex.Message}",
                        "Hiba!",
                        MessageBoxButtons.OK,
                        MessageBoxIcon.Error);
                }
            }            
        }        
    }
}

using System.ComponentModel;

namespace WinFormsApp1 {
    public partial class Form1 : Form {
        BindingList<string> listBoxItemsList = new BindingList<string>();

        public Form1() {
            InitializeComponent();
            listBox.DataSource = listBoxItemsList;
        }

        private void openToolStripMenuItem_Click(object sender, EventArgs e) {
            using OpenFileDialog ofd = new OpenFileDialog();

            ofd.Filter =
                "All Files (*.*)|*.*|" +
                "Text Files (*.txt)|*.txt|" +
                "PDF Files (*.pdf)|*.pdf|" +
                "CSV Files (*.csv)|*.csv";

            if (ofd.ShowDialog() == DialogResult.OK) {
                try {
                    string[] sorok = File.ReadAllLines(ofd.FileName);

                    foreach (var sor in sorok) {
                        if (!sor.Contains(";")) {
                            MessageBox.Show(
                                "A fájl hibás, legalább egy sor nem tartalmaz ';' karaktert",
                                "Hiba!",
                                MessageBoxButtons.OK,
                                MessageBoxIcon.Error);
                            return;
                        }
                    }

                    //Csak akkor törli az előző listát, ha sikerül beolvasni az újat és nem hiányzik ";" karakter
                    listBoxItemsList.Clear();

                    foreach (var sor in sorok) {
                        listBoxItemsList.Add(sor);
                    }
                } catch (Exception ex) {
                    MessageBox.Show(
                        $"Hiba történt: {ex.Message}",
                        "Hiba!",
                        MessageBoxButtons.OK,
                        MessageBoxIcon.Error);
                }
            }
        }      
    }
}

private void saveAsToolStripMenuItem_Click(object sender, EventArgs e) {
    using SaveFileDialog sfd = new SaveFileDialog();
    sfd.DefaultExt = "csv";
    sfd.AddExtension = true;

    if (sfd.ShowDialog() == DialogResult.OK) {
        try {
            File.WriteAllLines(sfd.FileName, listBoxItemsList);
            MessageBox.Show(
                "Sikeres mentés!",
                "Információ",
                MessageBoxButtons.OK,
                MessageBoxIcon.Information);
        } catch (Exception ex) {
            MessageBox.Show(
            "Hiba történt",
            "Hiba!",
            MessageBoxButtons.OK,
            MessageBoxIcon.Error);
        }
    }
}

private void addButton_Click(object sender, EventArgs e) {    
    bool arCheck = int.TryParse(arTextBox.Text, out int arValue);    

    if (!arCheck) {
        MessageBox.Show(
            "Az ár mező nem szám!",
            "Hiba",
            MessageBoxButtons.OK,
            MessageBoxIcon.Warning);
        return;
    }

    string data = $"{gyartoTextBox.Text};{modellTextBox.Text};{hatotavTextBox.Text};{repulesIdoTextBox.Text};{arTextBox.Text}";
    listBoxItemsList.Add(data);
    UpdateLineCountToolStripStatus();
}

////////// DEL BTN //////////
/////////////////////////////

private void deleteButton_Click(object sender, EventArgs e) {
    int selectedRow = listBox.SelectedIndex;

    if (selectedRow == -1) {
        MessageBox.Show(
            "Nincs kijelölt elem",
            "Figyelmeztetés",
            MessageBoxButtons.OK,
            MessageBoxIcon.Warning);
        return;
    }

    listBoxItemsList.RemoveAt(selectedRow);
    UpdateLineCountToolStripStatus();
}

private void updateButton_Click(object sender, EventArgs e) {
    int selectedRow = listBox.SelectedIndex;

    if (selectedRow == -1) {
        MessageBox.Show(
            "Nincs kijelölt elem",
            "Figyelmeztetés",
            MessageBoxButtons.OK,
            MessageBoxIcon.Warning);
        return;
    }
    
    bool arCheck = int.TryParse(arTextBox.Text, out int arValue);    

    if (!arCheck) {
        MessageBox.Show(
            "Az ár mező nem szám!",
            "Hiba",
            MessageBoxButtons.OK,
            MessageBoxIcon.Warning);
        return;
    }

    string newData = $"{gyartoTextBox.Text};{modellTextBox.Text};{hatotavTextBox.Text};{repulesIdoTextBox.Text};{arTextBox.Text}";
    listBoxItemsList[selectedRow] = newData;
}



//////////////////////////////
////////// SELECTED //////////
//////////////////////////////

private void listBox_SelectedIndexChanged(object sender, EventArgs e) {
    int selectedRow = listBox.SelectedIndex;

    if (selectedRow == -1) return;

    string value = listBoxItemsList[selectedRow];
    string[] values = value.Split(";");

    gyartoTextBox.Text = values[0].Trim();
    modellTextBox.Text = values[1].Trim();
    hatotavTextBox.Text = values[2].Trim();
    repulesIdoTextBox.Text = values[3].Trim();
    arTextBox.Text = values[4].Trim();
}
