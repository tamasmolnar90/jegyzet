pwd -- print working directory
kiírja a tartózkodási helyünket, hogy hol vagyunk a rendszerben

cd mappaneve -- change direcroty
belelép a mappába

cd ..
kilép a mappából

ls -- list
kilistázza a mappákat, ahol vagyunk
ls -a
rejtett fájlokat is kilistázza

touch filename
létrehoz egy file-t

echo szöveg > filename
beleírunk a fileba, ez mindig felülírja az előző szöveget

cat filename
file kiolvasása

nano filename
ctrl + o
enter
ctrl + x

nano filename.py
python3 filename.py

mkdir foldername -- make directory
létrehozunk egy mappát

mv filename foldername -- move
filenak és mappának egy mappában kell legyen

mv file1 /home/vision/Desktop/
fájl áthelyezése máshova teljes útvonallal

cp file1 file2 -- copy
file1 ből csinál egy másolatot file2 néven

cp file1.py /home/vision/Desktop/copyfile.py
úgy tudunk egy fájlt teljesen máshova másolni, ha megadjuk a teljes útvonalat

rm filename -- remove
kitöröl egy filet

rm foldername -r
kitöröl egy mappát

rm * -r
töröl minden a mappából amiben vagy

sudo

akkor csak sudo paranccsal tudjuk olvasni és szerkeszteni is.
