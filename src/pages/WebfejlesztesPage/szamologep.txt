const szamGombok = document.querySelectorAll(".szam");
const muveletGombok = document.querySelectorAll(".muvelet");
const egyenloGomb = document.querySelector("#egyenlo");
const torlesGomb = document.querySelector("#torles");
const kijelzo = document.querySelector("#kijelzo");

let elsoSzam;
let masodikSzam;
let kivalasztottMuvelet;

//Számgombok
szamGombok.forEach((gomb) => {
  gomb.addEventListener("click", () => {
    if (kijelzo.textContent === "0") {
      kijelzo.textContent = gomb.textContent;
    } else {
      kijelzo.textContent += gomb.textContent;
    }
  });
});

//Műveleti gombok
muveletGombok.forEach((gomb) => {
  gomb.addEventListener("click", () => {
    elsoSzam = parseFloat(kijelzo.textContent);
    kivalasztottMuvelet = gomb.textContent;
    kijelzo.textContent = "";
  });
});

//Egyenlőség
egyenloGomb.addEventListener("click", () => {
  masodikSzam = parseFloat(kijelzo.textContent);

  let eredmeny;

  if (kivalasztottMuvelet === "+") eredmeny = elsoSzam + masodikSzam;
  if (kivalasztottMuvelet === "-") eredmeny = elsoSzam - masodikSzam;
  if (kivalasztottMuvelet === "*") eredmeny = elsoSzam * masodikSzam;
  if (kivalasztottMuvelet === "/") eredmeny = elsoSzam / masodikSzam;

  kijelzo.textContent = eredmeny;
});

// Törlés
torlesGomb.addEventListener("click", () => {
  elsoSzam = "";
  masodikSzam = "";
  kivalasztottMuvelet = "";
  kijelzo.textContent = "0";
});

const szamGombok = document.querySelectorAll(".szam");
const muveletGombok = document.querySelectorAll(".muvelet");
const egyenloGomb = document.querySelector("#egyenlo");
const torlesGomb = document.querySelector("#torles");
const kijelzo = document.querySelector("#kijelzo");

let szamok;
let kivalasztottMuvelet;

//Számgombok
szamGombok.forEach((gomb) => {
  gomb.addEventListener("click", () => {
    if (kijelzo.textContent === "0") {
      kijelzo.textContent = gomb.textContent;
    } else {
      kijelzo.textContent += gomb.textContent;
    }
  });
});

//Műveleti gombok
muveletGombok.forEach((gomb) => {
  gomb.addEventListener("click", () => {
    szamok = parseFloat(kijelzo.textContent);
    kivalasztottMuvelet = gomb.textContent;
    kijelzo.textContent += " " + kivalasztottMuvelet + " ";
  });
});

//Egyenlőség
egyenloGomb.addEventListener("click", () => {
  let kijelzoReszek = kijelzo.textContent.split(kivalasztottMuvelet);
  let a = parseFloat(kijelzoReszek[0].trim());
  let b = parseFloat(kijelzoReszek[1].trim());

  let eredmeny;

  if (kivalasztottMuvelet === "+") eredmeny = a + b;
  if (kivalasztottMuvelet === "-") eredmeny = a - b;
  if (kivalasztottMuvelet === "*") eredmeny = a * b;
  if (kivalasztottMuvelet === "/") eredmeny = a / b;

  kijelzo.textContent = eredmeny;
});

// Törlés
torlesGomb.addEventListener("click", () => {
  szamok = "";
  kivalasztottMuvelet = "";
  kijelzo.textContent = "0";
});
