const inputEl = document.querySelector(".input");
const todoListEl = document.querySelector(".todo-list");
const addBtn = document.querySelector(".add-btn");
const modeBtn = document.querySelector(".mode-btn");

modeBtn.addEventListener("click", modeHandler);

addBtn.addEventListener("click", addTodoHandler);

todoListEl.addEventListener("click", delTodoHandler);

function modeHandler() {
  const isDark = modeBtn.textContent.trim() === "Dark-Mode";
  if (isDark) {
    document.body.style.backgroundColor = "#000";
    modeBtn.textContent = "White-Mode";
    modeBtn.style.backgroundColor = "#000";
    modeBtn.style.color = "#fff";
  } else {
    document.body.style.backgroundColor = "#fff";
    modeBtn.textContent = "Dark-Mode";
    modeBtn.style.backgroundColor = "#fff";
    modeBtn.style.color = "#000";
  }
}

function addTodoHandler() {
  const liEl = document.createElement("li");
  liEl.className = "todo-list-item";

  const pEl = document.createElement("p");
  pEl.className = "todo-list-item-text";
  pEl.textContent = inputEl.value;

  const delBtn = document.createElement("button");
  delBtn.className = "btn del-btn";
  delBtn.textContent = "DEL";

  //liEl.appendChild(pEl);
  //liEl.appendChild(delBtn);
  liEl.append(pEl, delBtn);
  todoListEl.append(liEl);
}

function delTodoHandler(e) {
  if (e.target.classList.contains("del-btn")) {
    e.target.closest(".todo-list-item").remove();
  }
}
