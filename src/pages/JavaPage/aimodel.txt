Mesterséges Intelligencia Modellek

A feladat egy olyan alkalmazás elkészítése, amely különböző AI gyártók modelljeit és azok specifikációit dolgozza fel.
A gyartok.csv UTF-8 kódolású forrásállomány sorai egy-egy modell adatait tartalmazzák, pontosvesszővel (;) elválasztva.

Adatszerkezet:
● Gyártó neve (pl. OpenAI)
● Modell neve (pl. GPT-4o)
● Ingyenes kérések száma naponta (egész szám vagy "NULL")
● Havi előfizetési díj USD-ben (egész szám vagy "NULL")
● Kontextus ablak mérete tokenekben (egész szám)

Feladatok:
1. Hozzon létre egy projektet AIModellek néven!
2. Olvassa be a gyartok.csv állomány adatait, és tárolja egy összetett adatszerkezetben, amelyet egy AIModel nevű osztály reprezentál.
   Ügyeljen a "NULL" értékek kezelésére (használhat 0-t vagy nullable típust)!
3. Számolja meg és írassa ki, összesen hány modell adatai szerepelnek a fájlban!
4. Keresse meg azt a modellt, amely a legnagyobb kontextus ablakkal rendelkezik, és írassa ki a modell nevét és a gyártóját!
5. Keresse meg azt a modellt, amelynek a legolcsóbb a havi előfizetése (a NULL / 0 értékeket hagyja figyelmen kívül, csak a fizetős modelleket nézze)!
6. Számolja meg, hány olyan modell van a listában, amelynek a gyártója a "Google"!
7. Számolja ki az összes modell átlagos kontextus ablak méretét, és írassa ki egész számra kerekítve!
8. Készítsen egy listát a "Mistral AI" gyártó modelljeiről, és írassa ki csak a modellneveket a képernyőre!

public class AIModel {
    private String gyartoNeve;
    private String modellNeve;
    private Integer napiLimit;
    private Integer haviElofizetes;
    private int kontextusAblak;

    public AIModel(String gyartoNeve, String modellNeve, Integer napiLimit, Integer haviElofizetes, int kontextusAblak) {
        this.gyartoNeve = gyartoNeve;
        this.modellNeve = modellNeve;
        this.napiLimit = napiLimit;
        this.haviElofizetes = haviElofizetes;
        this.kontextusAblak = kontextusAblak;
    }

    public String getGyartoNeve() {
        return gyartoNeve;
    }

    public void setGyartoNeve(String gyartoNeve) {
        this.gyartoNeve = gyartoNeve;
    }

    public String getModellNeve() {
        return modellNeve;
    }

    public void setModellNeve(String modellNeve) {
        this.modellNeve = modellNeve;
    }

    public Integer getNapiLimit() {
        return napiLimit;
    }

    public void setNapiLimit(Integer napiLimit) {
        this.napiLimit = napiLimit;
    }

    public Integer getHaviElofizetes() {
        return haviElofizetes;
    }

    public void setHaviElofizetes(Integer haviElofizetes) {
        this.haviElofizetes = haviElofizetes;
    }

    public int getKontextusAblak() {
        return kontextusAblak;
    }

    public void setKontextusAblak(int kontextusAblak) {
        this.kontextusAblak = kontextusAblak;
    }
}

import java.io.IOException;
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.ArrayList;
import java.util.List;

public class Main {

    public static void main(String[] args) throws IOException {

        List<String> gyartokAdat = Files.readAllLines(Path.of("gyartok.csv"));
        String header = gyartokAdat.get(0);
        //String header = gyartokAdat.getFirst();

        List<AIModel> gyartokLista = new ArrayList<AIModel>();

        for (int i = 1; i < gyartokAdat.size(); i++) {
            String[] adat = gyartokAdat.get(i).split(";");

            AIModel aiModel = new AIModel(
                    adat[0],
                    adat[1],
                    parseNullableInt(adat[2]),
                    parseNullableInt(adat[3]),
                    Integer.parseInt(adat[4])
            );

            gyartokLista.add(aiModel);
        }

        //List<AIModel> gyartokLista = gyartokAdat.stream()
        //        .skip(1)
        //        .map(sor->sor.split(";"))
        //        .map(adat -> new AIModel(
        //                adat[0],
        //                adat[1],
        //                parseNullableInt(adat[2]),
        //                parseNullableInt(adat[3]),
        //                Integer.parseInt(adat[4])
        //        )).toList();

        //Összes sor kiíratása
        //for(AIModel sor : gyartokLista){
        //    System.out.println(String.format("%s, %s, %d, %d, %d",
        //            sor.getGyartoNeve(),
        //            sor.getModellNeve(),
        //            sor.getNapiLimit(),
        //            sor.getHaviElofizetes(),
        //            sor.getKontextusAblak()));
        //}

        //03. Számolja meg és írassa ki, összesen hány modell adatai szerepelnek a fájlban!
        int modellekSzama = gyartokLista.size();
        System.out.printf("3. feladat: Modellek száma: %s db", modellekSzama);

        //04. Keresse meg azt a modellt, amely a legnagyobb kontextus ablakkal rendelkezik, és írassa ki a modell nevét és a gyártóját!
        AIModel legnagyobbKontextus = null;

        for (int i = 0; i < gyartokLista.size(); i++) {
            if (legnagyobbKontextus == null || gyartokLista.get(i).getKontextusAblak() > legnagyobbKontextus.getKontextusAblak()){
                legnagyobbKontextus = gyartokLista.get(i);
            }
        }

        System.out.printf("\n4. feladat: Legnagyobb kontextus: %s - %s (%d token)",
                legnagyobbKontextus.getGyartoNeve(),
                legnagyobbKontextus.getModellNeve(),
                legnagyobbKontextus.getKontextusAblak());

        //05. Keresse meg azt a modellt, amelynek a legolcsóbb a havi előfizetése (a NULL / 0 értékeket hagyja figyelmen kívül, csak a fizetős modelleket nézze)!
        AIModel legolcsobbElofizetes = null;

        for (int i = 0; i < gyartokLista.size() ; i++) {
            if (gyartokLista.get(i).getHaviElofizetes() == null) {
                continue;
            }

            if (legolcsobbElofizetes == null || gyartokLista.get(i).getHaviElofizetes() < legolcsobbElofizetes.getHaviElofizetes()){
                legolcsobbElofizetes = gyartokLista.get(i);
            }
        }

        System.out.printf("\n5. feladat: Legolcsóbb fizetős modell: %s (%d USD)",
                legolcsobbElofizetes.getModellNeve(),
                legolcsobbElofizetes.getHaviElofizetes());

        //06. Számolja meg, hány olyan modell van a listában, amelynek a gyártója a "Google"!
        int googleModellekSzama = 0;

        for (int i = 0; i < gyartokLista.size(); i++){
            if(gyartokLista.get(i).getGyartoNeve().contains("Google")){
                googleModellekSzama++;
            }
        }
        System.out.printf("\n6. feladat: Google modellek száma: %d db", googleModellekSzama);

        //07. Számolja ki az összes modell átlagos kontextus ablak méretét, és írassa ki egész számra kerekítve!
        int osszesKontextus = 0;
        int darabKontextus = 0;

        for (int i = 0; i < gyartokLista.size(); i++){
            osszesKontextus += gyartokLista.get(i).getKontextusAblak();
            darabKontextus++;
        }

        float atlagKontextus = (float) osszesKontextus / darabKontextus;

        System.out.printf("\n7. feladat: Átlagos kontextus méret: %d token", Math.round(atlagKontextus));
        //System.out.printf("\n7. feladat: Átlagos kontextus méret: %.0f token", atlagKontextus);

        //08. Készítsen egy listát a "Mistral AI" gyártó modelljeiről, és írassa ki csak a modellneveket a képernyőre!
        List<AIModel> mistralModellek = new ArrayList<AIModel>();

        for (int i = 0; i < gyartokLista.size() ; i++) {
            if(gyartokLista.get(i).getGyartoNeve().contains("Mistral AI")){
                mistralModellek.add(gyartokLista.get(i));
            }
        }

        System.out.print("8. feladat: Mistral AI modellek:");

        for (int i = 0; i < mistralModellek.size(); i++) {
            System.out.printf("\n\t- %s", mistralModellek.get(i).getModellNeve());
        }
    }

    //HELPER METHOD
    static Integer parseNullableInt(String value) {
        if (value == null) return null;

        value = value.trim();

        if (value.isEmpty() || value.equalsIgnoreCase("null")) return null;

        try {
            return Integer.parseInt(value);
        } catch (NumberFormatException e) {
            return null;
        }
    }
}
