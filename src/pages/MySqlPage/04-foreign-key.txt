CREATE TABLE szerzo (
    id INT PRIMARY KEY AUTO_INCREMENT,
    nev VARCHAR(255) NOT NULL
);

CREATE TABLE konyv (
    id INT PRIMARY KEY AUTO_INCREMENT,
    cim VARCHAR(255) NOT NULL,
    szerzo_id INT NOT NULL
);

ALTER TABLE konyv
ADD CONSTRAINT fk_konyv_szerzo
FOREIGN KEY(szerzo_id)
REFERENCES szerzo(id)
ON DELETE RESTRICT
ON UPDATE CASCADE;



ON DELETE CASCADE → ha törlöd a szerzőt, a könyvei is törlődnek.
ON UPDATE CASCADE → ha megváltozik a szerző id-je, a kapcsolódó könyvekben is frissül.
Ha már vannak adatok a konyv táblában, akkor a foreign key létrehozása csak akkor sikerül, ha minden szerzo_id valóban létezik a szerzo.id oszlopban. Ellenkező esetben MySQL hibát fog dobni.

ON DELETE RESTRICT → ha megpróbálod törölni a szerzőt, és van hozzá kapcsolódó könyv, a törlés nem fog megtörténni (hibát kapsz). A rendszer megakadályozza az „árva” könyvek létrejöttét.
ON UPDATE RESTRICT → ha megpróbálod módosítani a szerző id értékét, és erre az id-re hivatkozik könyv, az UPDATE nem fog lefutni (hibát kapsz). Az adatok konzisztenciája miatt a hivatkozott kulcs nem változtatható.
Ha már vannak adatok a konyv táblában, akkor a foreign key létrehozása csak akkor sikerül, ha minden szerzo_id valóban létezik a szerzo.id oszlopban. Ellenkező esetben MySQL hibát fog dobni, és a constraint nem jön létre.



CREATE TABLE szerzo (
    id INT PRIMARY KEY AUTO_INCREMENT,
    nev VARCHAR(255) NOT NULL
);

CREATE TABLE konyv (
    id INT PRIMARY KEY AUTO_INCREMENT,
    cim VARCHAR(255) NOT NULL,
    szerzo_id INT NOT NULL,

    FOREIGN KEY(szerzo_id)
    REFERENCES szerzo(id)
    ON DELETE RESTRICT
    ON UPDATE CASCADE
);
