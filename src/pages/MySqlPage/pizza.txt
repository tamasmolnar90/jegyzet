-- 05. Ki szállította házhoz az első (egyes sorszámú) rendelést?
SELECT futar.futar_nev
FROM rendeles
INNER JOIN futar ON rendeles.futar_id = futar.futar_id
WHERE rendeles.rendeles_id = 1;



-- 06. Kik rendeltek pizzát délelőtt?
SELECT DISTINCT vevo.vevo_nev
FROM rendeles
INNER JOIN vevo ON rendeles.vevo_id = vevo.vevo_id
WHERE TIME(rendeles.rendeles_idopont) < '12:00:00';



-- 07. Milyen pizzákat evett Morgó?
SELECT p.pizza_nev, t.tetel_db
FROM vevo v
JOIN rendeles r ON v.vevo_id = r.vevo_id
JOIN tetel t ON r.rendeles_id = t.rendeles_id
JOIN pizza p ON t.pizza_id = p.pizza_id
WHERE v.vevo_nev = 'Morgó';



-- 08. Ki szállított házhoz Tudornak?
SELECT DISTINCT f.futar_nev
FROM rendeles r
JOIN futar f ON r.futar_id = f.futar_id
JOIN vevo v ON r.vevo_id = v.vevo_id
WHERE v.vevo_nev = 'Tudor';




-- 09. Az egyes rendelések alkalmával ki kinek szállított házhoz?
SELECT r.rendeles_id, v.vevo_nev AS vevo, f.futar_nev AS futar, r.rendeles_idopont
FROM rendeles r
JOIN vevo v ON r.vevo_id = v.vevo_id
JOIN futar f ON r.futar_id = f.futar_id
ORDER BY r.rendeles_id;



-- 10. Mennyit költött pizzára Vidor?
SELECT v.vevo_nev, COALESCE(SUM(t.tetel_db * p.pizza_ar),0) AS ossz_koltseg
FROM vevo v
LEFT JOIN rendeles r ON v.vevo_id = r.vevo_id
LEFT JOIN tetel t ON r.rendeles_id = t.rendeles_id
LEFT JOIN pizza p ON t.pizza_id = p.pizza_id
WHERE v.vevo_nev = 'Vidor'
GROUP BY v.vevo_id;



-- 11. Hány alkalommal rendelt Sorrento pizzát Kuka?
SELECT COUNT(DISTINCT r.rendeles_id) AS alkalom
FROM rendeles r
JOIN tetel t ON r.rendeles_id = t.rendeles_id
JOIN pizza p ON t.pizza_id = p.pizza_id
JOIN vevo v ON r.vevo_id = v.vevo_id
WHERE v.vevo_nev = 'Kuka' AND p.pizza_nev = 'Sorrento';

-- 21. Mennyi pizza fogyott átlagosan naponta?
SELECT AVG(napi_db) AS atlag_napi_fogyasztas
FROM (
  SELECT DATE(r.rendeles_idopont) AS nap, SUM(t.tetel_db) AS napi_db
  FROM rendeles r
  JOIN tetel t ON r.rendeles_id = t.rendeles_id
  GROUP BY nap
) AS napi;



-- 22. Hány pizzát rendeltek átlagosan egyszerre?
SELECT AVG(order_db) AS atlag_db_rendelesenkent
FROM (
  SELECT r.rendeles_id, SUM(t.tetel_db) AS order_db
  FROM rendeles r
  JOIN tetel t ON r.rendeles_id = t.rendeles_id
  GROUP BY r.rendeles_id

-- 35. Mely pizzák olcsóbbak a Capricciosa pizzánál?

);



-- 36. Mely pizzák drágábbak az átlagosnál?
SELECT pizza_nev, pizza_ar
FROM pizza
WHERE pizza_ar > (SELECT AVG(pizza_ar) FROM pizza);



-- 37. Mely pizza ára van legközelebb az átlagárhoz?
SELECT pizza_nev, pizza_ar, ABS(pizza_ar - avg_ar) AS diff
FROM (
  SELECT p.*, (SELECT AVG(pizza_ar) FROM pizza) AS avg_ar
  FROM pizza p
) AS sub
ORDER BY diff
LIMIT 1;



-- 38. Mely futárok mentek többet házhoz az átlagosnál?
WITH futar_totals AS (
  SELECT f.futar_id, f.futar_nev, COALESCE(SUM(t.tetel_db),0) AS total
  FROM futar f
  LEFT JOIN rendeles r ON f.futar_id = r.futar_id
  LEFT JOIN tetel t ON r.rendeles_id = t.rendeles_id
  GROUP BY f.futar_id
)
SELECT ft.futar_nev, ft.total
FROM futar_totals ft
WHERE ft.total > (SELECT AVG(total) FROM futar_totals);



-- 39. Kik rendeltek legalább háromszor annyi pizzát, mint egy átlagos vevő?
WITH vevo_totals AS (
  SELECT v.vevo_id, v.vevo_nev, COALESCE(SUM(t.tetel_db),0) AS total
  FROM vevo v
  LEFT JOIN rendeles r ON v.vevo_id = r.vevo_id
  LEFT JOIN tetel t ON r.rendeles_id = t.rendeles_id
  GROUP BY v.vevo_id
)
SELECT vt.vevo_nev, vt.total
FROM vevo_totals vt
WHERE vt.total >= 3 * (SELECT AVG(total) FROM vevo_totals);



-- 40. Kik szállítottak házhoz legalább tízszer?
SELECT f.futar_nev, COUNT(r.rendeles_id) AS rendeles_szam
FROM futar f
LEFT JOIN rendeles r ON f.futar_id = r.futar_id
GROUP BY f.futar_id
HAVING rendeles_szam >= 10;



-- 41. Mely pizzából fogyott legalább 50 db?

GROUP BY p.pizza_id

-- 42. Mely vevők nem rendeltek legalább háromszor?

-- 43. Kik rendeltek legalább 5 Hawaii pizzát?

-- 44. Milyen pizzából nem rendelt soha Tudor?
SELECT p.pizza_nev
FROM pizza p
WHERE p.pizza_id NOT IN (
  SELECT t.pizza_id
  FROM tetel t
  JOIN rendeles r ON t.rendeles_id = r.rendeles_id
  JOIN vevo v ON r.vevo_id = v.vevo_id
  WHERE v.vevo_nev = 'Tudor'
);



-- 45. Van-e olyan pizza, amelyből soha nem rendeltek?
SELECT p.pizza_nev
FROM pizza p
LEFT JOIN tetel t ON p.pizza_id = t.pizza_id
WHERE t.pizza_id IS NULL;



-- 46. Ki nem rendelt soha Vesuvio pizzát?
SELECT v.vevo_nev
FROM vevo v
WHERE v.vevo_id NOT IN (
  SELECT r.vevo_id
  FROM rendeles r
  JOIN tetel t ON r.rendeles_id = t.rendeles_id
  JOIN pizza p ON t.pizza_id = p.pizza_id
  WHERE p.pizza_nev = 'Vesuvio'
);



-- 47. Mely pizzafutárokkal nem találkoztak az egyes vevők?
SELECT v.vevo_nev, f.futar_nev
FROM vevo v
CROSS JOIN futar f
LEFT JOIN rendeles r ON r.vevo_id = v.vevo_id AND r.futar_id = f.futar_id
WHERE r.rendeles_id IS NULL
ORDER BY v.vevo_nev, f.futar_nev;



-- 48. Kik rendeltek több Sorrento pizzát, mint Vesuviot?
SELECT v.vevo_nev,
  SUM(CASE WHEN p.pizza_nev = 'Sorrento' THEN t.tetel_db ELSE 0 END) AS sorrento_db,
  SUM(CASE WHEN p.pizza_nev = 'Vesuvio' THEN t.tetel_db ELSE 0 END) AS vesuvio_db
FROM vevo v
LEFT JOIN rendeles r ON v.vevo_id = r.vevo_id
LEFT JOIN tetel t ON r.rendeles_id = t.rendeles_id
LEFT JOIN pizza p ON t.pizza_id = p.pizza_id
GROUP BY v.vevo_id
HAVING sorrento_db > vesuvio_db;



-- 49. Kik rendeltek legalább 5 Capricciosa vagy 8 Frutti di Mare pizzát?
SELECT v.vevo_nev,
  SUM(CASE WHEN p.pizza_nev = 'Capricciosa' THEN t.tetel_db ELSE 0 END) AS capri_db,
  SUM(CASE WHEN p.pizza_nev = 'Frutti di Mare' THEN t.tetel_db ELSE 0 END) AS frutti_db
FROM vevo v
LEFT JOIN rendeles r ON v.vevo_id = r.vevo_id
LEFT JOIN tetel t ON r.rendeles_id = t.rendeles_id
LEFT JOIN pizza p ON t.pizza_id = p.pizza_id
GROUP BY v.vevo_id
HAVING capri_db >= 5 OR frutti_db >= 8;



-- 50. Kik rendeltek kétfajta pizzából is legalább 10 darabot?
SELECT v.vevo_nev
FROM vevo v
JOIN (
  SELECT r.vevo_id, t.pizza_id, SUM(t.tetel_db) AS db
  FROM rendeles r
  JOIN tetel t ON r.rendeles_id = t.rendeles_id
  GROUP BY r.vevo_id, t.pizza_id
) AS vp ON v.vevo_id = vp.vevo_id
GROUP BY v.vevo_id
HAVING SUM(CASE WHEN vp.db >= 10 THEN 1 ELSE 0 END) >= 2;
