DROP DATABASE IF EXISTS taxi;
CREATE DATABASE taxi
  CHARACTER SET utf8mb4
  COLLATE utf8mb4_hungarian_ci;

USE taxi;

CREATE TABLE SOFOR (
  rendszam VARCHAR(8) NOT NULL,
  nev VARCHAR(30) DEFAULT NULL,
  szuldatum DATE DEFAULT NULL,
  PRIMARY KEY (rendszam)
) ENGINE=InnoDB
  DEFAULT CHARSET=utf8mb4
  COLLATE=utf8mb4_hungarian_ci;

CREATE TABLE TARIFA (
  tarifakod INT NOT NULL,
  megnevezes VARCHAR(30) NOT NULL,
  dij INT DEFAULT NULL,
  PRIMARY KEY (tarifakod)
) ENGINE=InnoDB
  DEFAULT CHARSET=utf8mb4
  COLLATE=utf8mb4_hungarian_ci;

CREATE TABLE FUVAR (
  rendszam VARCHAR(8) NOT NULL,
  indul DATETIME NOT NULL,
  erkezik DATETIME DEFAULT NULL,
  km INT DEFAULT NULL,
  tarifa INT DEFAULT NULL,

  PRIMARY KEY (rendszam, indul),

  KEY idx_fuvar_tarifa (tarifa),

  CONSTRAINT fk_fuvar_tarifa
    FOREIGN KEY (tarifa)
    REFERENCES TARIFA (tarifakod)
    ON DELETE SET NULL
    ON UPDATE CASCADE,

  CONSTRAINT fk_fuvar_sofor
    FOREIGN KEY (rendszam)
    REFERENCES SOFOR (rendszam)
    ON DELETE NO ACTION
    ON UPDATE CASCADE
) ENGINE=InnoDB
  DEFAULT CHARSET=utf8mb4
  COLLATE=utf8mb4_hungarian_ci;
