-- 01. Listázd az összes gép nevét és típusát!
SELECT nev, tipus
FROM gep;




-- 02. Gépek, amelyeket 2019 után helyeztek üzembe!
SELECT *
FROM gep
WHERE uzembe_helyezes > 2019;




-- 03. Tatabányán dolgozó operátorok!
SELECT *
FROM operator
WHERE telephely = 'Tatabánya';




-- 04. Karbantartások, ahol a költség 100000 Ft fölött volt!
SELECT
  gep.nev,
  gep.tipus,
  karbantartas.datum AS karbantartas_datum,
  karbantartas.koltseg,
  karbantartas.megjegyzes
FROM karbantartas
JOIN gep ON karbantartas.gep_id = gep.id
WHERE karbantartas.koltseg > 100000;




-- 05. NINCS ILYEN FELADAT A BEADANDÓK KÖZÖTT!!!




-- 06. Gépek ár szerint csökkenő sorrendben!
SELECT *
FROM gep
ORDER BY ar DESC;




-- 07. A három legdrágább gép!
SELECT *
FROM gep
ORDER BY ar DESC
LIMIT 3;




-- 08. Operátorok név szerint növekvő sorrendben!
SELECT *
FROM operator
ORDER BY nev ASC;




-- 09. Telephelyenkénti átlagbérek!
SELECT telephely, AVG(ber) AS atlagber
FROM operator
GROUP BY telephely;




-- 10. A 2020 előtt üzembe helyezett gépek száma!
SELECT COUNT(*) AS gepek_szama
FROM gep
WHERE uzembe_helyezes < 2020;




-- 11. Gépek átlagára!
SELECT AVG(ar) AS gepek_atlagara
FROM gep;




-- 12. Operátorok neve és névhossz!
SELECT nev, CHAR_LENGTH(nev) AS nevhossz
FROM operator;




-- 13. Karbantartások összköltsége!
SELECT SUM(koltseg) AS osszkoltseg
FROM karbantartas;




-- 14. 2024-ben történt karbantartások!
SELECT *
FROM karbantartas
WHERE datum BETWEEN '2024-01-01' AND '2024-12-31';




-- 15. Karbantartók bérének növelése 10%-kal!
SET SQL_SAFE_UPDATES = 0;

UPDATE operator
SET ber = ber * 1.10
WHERE beosztas = 'Karbantartó';

SET SQL_SAFE_UPDATES = 1;
