Beállítások

git config --global user.name //lekérdezés
git config --global user.name "Teljes Név" //beállítás
git config --global --unset user.name //töröl

git config --global user.email
git config --global user.email "example@gmail.com"
git config --global --unset user.email

git config --global --remove-section user //eltávolítja a nevet, emailt
git config --global --unset-all safe.directory
git config --global --remove-section filter.lfs

del %USERPROFILE%\.gitconfig //eltávolít mindent

git config --global –list
