beker = int(input("Adj meg egy számot: "))
def faktorialis(n):
    if n < 0:
        return "A faktoriális nem értelmezett negatív számra!"
    elif n == 0:
        return 1
    else:
        return n * faktorialis(n - 1)
print(f"A(z) {beker} faktoriálisa: {faktorialis(beker)}")

### 18 | Kérj be 10db számot, írj függvényt, ami paraméterben várja a listát és adja vissza a legkisebb számot és annak indexét.

    beker = int(input(f"Adja meg a(z) {i+1}. számot: "))
    lista.append(beker)

def min_number_index(li):
    min_num = li[0]
    min_index = 0

    for i in range(len(li)):
        if li[i] < min_num:
            min_num = li[i]
            min_index = i

    return min_num, min_index

min_num, min_index = min_number_index(lista)
