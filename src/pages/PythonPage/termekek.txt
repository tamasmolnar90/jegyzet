import os


class Termek:
    def __init__(self, termek_id, megnevezes, kategoria, ar, keszlet):
        self.termek_id = termek_id
        self.megnevezes = megnevezes
        self.kategoria = kategoria
        self.ar = ar
        self.keszlet = keszlet


termek_lista = []

# Használd a file relatív útvonalát a script helyéhez képest!
# Így biztosan az adott mappa fájlját adja meg, függetlenül a current working directory-től.

        termek = Termek(int(adat[0]), adat[1], adat[2], float(adat[3]), int(adat[4]))
        termek_lista.append(termek)
