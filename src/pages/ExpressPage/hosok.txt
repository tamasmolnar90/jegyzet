import express from "express";
import mysql from "mysql2/promise";

const pool = mysql.createPool({
  host: "localhost",
  port: 3306,
  user: "root",
  password: "mysql123",
  database: "heroes_db",
});

const app = express();
app.use(express.json());

// GET
app.get("/api/hosok", async (_req, res) => {
  try {
    const [rows] = await pool.execute(
      `
        SELECT
          hos.id,
          hos.nev,
          hos.szarmazas,
          hos.szint,
          kaszt.id AS kaszt_id,
          kaszt.nev AS kaszt_nev
        FROM hos
        INNER JOIN kaszt ON hos.kaszt_id = kaszt.id
      `,
    );

    res.status(200).json(
      rows.map((hos) => ({
        id: hos.id,
        nev: hos.nev,
        szarmazas: hos.szarmazas,
        szint: hos.szint,
        kaszt: {
          id: hos.kaszt_id,
          nev: hos.kaszt_nev,
        },
      })),
    );

    // const [rows] = await pool.execute(

// POST
app.post("/api/hosok", async (req, res) => {
  try {
    const { nev, szarmazas, szint, kaszt_id } = req.body;

    if (!nev || !szarmazas || szint == null || kaszt_id == null) {
      return res.status(400).json({
        message:
          "Hiányzó adatok: név, származás, szint és kaszt megadása kötelező.",
      });
    }

    const [hosResult] = await pool.execute(
      `
        INSERT INTO hos (nev, szarmazas, szint, kaszt_id)
        VALUES (?, ?, ?, ?)        
      `,
      [nev, szarmazas, szint, kaszt_id],
    );

    const hosId = hosResult.insertId;

    const [rows] = await pool.execute(
      `
        SELECT id, nev, szarmazas, szint, kaszt_id
        FROM hos
        WHERE id = ?
      `,
      [hosId],
    );

    res.status(201).json({
      message: "Sikeresen hozzáadva",
      hos: rows[0],
    });
  } catch (error) {
    console.error(error);
    res.status(500).json(error);
  }
});

// GET
app.get("/api/kasztok", async (_req, res) => {
  try {
    const [rows] = await pool.execute("SELECT * FROM kaszt");

    res.status(200).json(rows);
  } catch (error) {
    console.error(error);
    res.status(500).json(error);
  }
});

// DELETE
app.delete("/api/hosok/:id", async (req, res) => {
  try {
    const { id } = req.params;

    const [result] = await pool.execute("DELETE FROM hos WHERE id = ?", [id]);

    if (result.affectedRows === 0) {
      res.sendStatus(404);
    } else {
      res.sendStatus(204);
    }
  } catch (error) {
    console.error(error);
    res.status(500).json(error);
  }
});

const PORT = 5000;
app.listen(PORT, () =>
  console.log(`Server is running: http://localhost:${PORT}\n`),
);
