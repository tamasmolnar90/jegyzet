CREATE DATABASE IF NOT EXISTS PROGRAMOZASI_NYELVEK
CHARACTER SET utf8mb4
COLLATE utf8mb4_hungarian_ci;

SET SQL_SAFE_UPDATES = 0;

USE PROGRAMOZASI_NYELVEK;

CREATE TABLE programozasi_nyelvek (
  nyelv_id INT PRIMARY KEY AUTO_INCREMENT,
  nev VARCHAR(50) NOT NULL,
  paradigma VARCHAR(100),
  megjelenes_eve INT
);

CREATE TABLE keretrendszerek (
  keretrendszer_id INT PRIMARY KEY AUTO_INCREMENT,
  nyelv_id INT,
  nev VARCHAR(100),
  celterulet VARCHAR(100),
  aktualis_verzio VARCHAR(50)
);

ALTER TABLE keretrendszerek
ADD CONSTRAINT fk_keretr_progny
FOREIGN KEY(nyelv_id)
REFERENCES programozasi_nyelvek(nyelv_id)
ON DELETE RESTRICT
ON UPDATE CASCADE;

import express from "express";
import mysql from "mysql2/promise";

const pool = mysql.createPool({
  host: "localhost",
  port: 3306,
  user: "root",
  password: "mysql123",
  database: "PROGRAMOZASI_NYELVEK",
});

const app = express();

app.use(express.json());

app.get("/api/v1/keretrendszerek", async (_req, res) => {
  try {
    const [rows] = await pool.query(
      `
       SELECT
        keretrendszerek.keretrendszer_id,
        keretrendszerek.nev,
        keretrendszerek.celterulet,
        keretrendszerek.aktualis_verzio,
        keretrendszerek.nyelv_id,
        JSON_OBJECT(
          'nyelv_id', programozasi_nyelvek.nyelv_id,
          'nev', programozasi_nyelvek.nev,
          'paradigma', programozasi_nyelvek.paradigma,
          'megjelenes_eve', programozasi_nyelvek.megjelenes_eve
        ) AS progny
        FROM keretrendszerek
        INNER JOIN programozasi_nyelvek
          ON keretrendszerek.nyelv_id = programozasi_nyelvek.nyelv_id
      `,
    );
    res.status(200).json(rows);
  } catch (error) {
    console.error(error);
    res.status(500).json(error);
  }
});

app.post("/api/v1/keretrendszerek", async (req, res) => {
  const { nyelv_id, nev, celterulet, aktualis_verzio } = req.body;

  if (!nev || !nyelv_id) {
    return res.status(400).json({
      üzenet: "Hiányzó adatok, a név és a nyelv_id kötelezőek!",
    });
  }

  try {
    const [result] = await pool.execute(
      `
      INSERT INTO keretrendszerek (nyelv_id, nev, celterulet, aktualis_verzio)
      VALUES (?, ?, ?, ?)
      `,
      [nyelv_id, nev, celterulet, aktualis_verzio],
    );

    res.status(200).json({
      üzenet: "Sikeres adatfelvétel!",
    });
  } catch (error) {
    console.error(error);
    res.status(500).json({
      üzenet: "Sikertelen adatfelvétel!",
    });
  }
});

app.patch("/api/v1/keretrendszerek", async (req, res) => {
  const { id, celterulet, aktualis_verzio } = req.body;

  try {
    const [result] = await pool.execute(
      `
        UPDATE keretrendszerek
        SET
          celterulet = ?,
          aktualis_verzio = ?
        WHERE nyelv_id = ?
      `,
      [celterulet, aktualis_verzio, id],
    );

    if (result.affectedRows === 0) {
      return res.sendStatus(400);
    }

    res.sendStatus(200);
  } catch (error) {
    console.error(error);
  }
});

app.delete("/api/v1/keretrendszerek/:id", async (req, res) => {
  const { id } = req.params;
  const keretrendszerId = Number(id);

  try {
    const [result] = await pool.execute(
      `
        DELETE
        FROM keretrendszerek
        WHERE keretrendszer_id = ?
      `,
      [keretrendszerId],
    );

    if (result.affectedRows === 0) {
      return res.sendStatus(404);
    }

    res.sendStatus(204);
  } catch (error) {
    console.error(error);
  }
});

const PORT = 3300;
app.listen(PORT, () => {
  console.log(`Server is running: http://localhost:${PORT}\n`);
});
