CREATE DATABASE AI_ADATOK
CHARACTER SET utf8mb4
COLLATE utf8mb4_hungarian_ci;

SET SQL_SAFE_UPDATES = 0;

USE AI_ADATOK;

CREATE TABLE gyartok(
  gyarto_id INT PRIMARY KEY AUTO_INCREMENT,
  nev VARCHAR(100) NOT NULL,
  weboldal VARCHAR(255)
);

CREATE TABLE ai_modellek(
  modell_id INT PRIMARY KEY AUTO_INCREMENT,
  gyarto_id INT NOT NULL,
  nev VARCHAR(100) NOT NULL,
  napi_lekerdezes_limit INT,
  percenkenti_lekerdezes_limit INT,
  kontextus_ablak INT
);

ALTER TABLE ai_modellek
ADD CONSTRAINT fk_ai_modellek_gyartok
FOREIGN KEY(gyarto_id)
REFERENCES gyartok(gyarto_id)
ON DELETE RESTRICT
ON UPDATE CASCADE;

app.post("/api/v1/ai-model", async (req, res) => {
  try {
    const {
      nev,
      napi_lekerdezes_limit,
      percenkenti_lekerdezes_limit,
      kontextus_ablak,
      gyarto_id,
    } = req.body;

    if (!nev || !gyarto_id) {
      return res.status(400).json({
        üzenet: "Hiányzó adatok, a név és a gyarto_id kötelezőek!",
      });
    }

    await pool.execute(
      `
        INSERT INTO ai_modellek (
          nev,
          napi_lekerdezes_limit,
          percenkenti_lekerdezes_limit,
          kontextus_ablak,
          gyarto_id
        )
        VALUES (?, ?, ?, ?, ?)
      `,
      [
        nev,
        napi_lekerdezes_limit,
        percenkenti_lekerdezes_limit,
        kontextus_ablak,
        gyarto_id,
      ],
    );

    // adatok felvételénél 201 a helyes, de a feladat 200-at kért
    res.status(200).json({
      üzenet: "Sikeres adatfelvétel!",
    });
  } catch (error) {
    console.error(error);
    res.status(500).json("Sikertelen adatfelvétel!");
  }
});

app.patch("/api/v1/ai-model", async (req, res) => {
  try {
    const {
      modell_id,
      napi_lekerdezes_limit,
      percenkenti_lekerdezes_limit,
      kontextus_ablak,
    } = req.body;

    if (!modell_id) {
      return res.status(400).json("Hiányzó modell_id");
    }

    const [result] = await pool.execute(
      `
        UPDATE ai_modellek
        SET
          napi_lekerdezes_limit = ?,
          percenkenti_lekerdezes_limit = ?,
          kontextus_ablak = ?
        WHERE modell_id = ?
      `,
      [
        napi_lekerdezes_limit,
        percenkenti_lekerdezes_limit,
        kontextus_ablak,
        modell_id,
      ],
    );

    if (result.affectedRows == 0) {
      res.sendStatus(400);
    } else {
      res.sendStatus(200);
    }
  } catch (error) {
    console.error(error);
    res.status(500).json(error);
  }
});

app.delete("/api/v1/ai-model/:id", async (req, res) => {
  try {
    const { id } = req.params;

    const [result] = await pool.execute(
      `
        DELETE FROM ai_modellek
        WHERE modell_id = ?
      `,
      [id],
    );

    if (result.affectedRows == 0) {
      res.sendStatus(404);
    } else {
      res.sendStatus(204);
    }
  } catch (error) {
    console.error(error);
    res.status(500).json(error);
  }
});

const PORT = 3300;
app.listen(PORT, () => {
