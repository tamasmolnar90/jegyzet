-- 01. Számold meg, hány darab aktív eszköz van minden típusból!
SELECT type, COUNT(*) AS aktiv_eszkozok_szama
FROM devices
WHERE status = 'active'
GROUP BY type;

-- 02. Módosítsd az összes "inactive" eszközt "active"-ra!
UPDATE devices
SET status = 'active'
WHERE status = 'inactive';

-- 03. Töröld azokat a szenzoradatokat, amelyek egy hónapnál régebbiek!
DELETE FROM sensor_data
WHERE recorded_at < NOW() - INTERVAL '1 month';

-- 04. Listázd ki azokat a felhasználókat, akik rendelkeznek "admin" szerepkörrel!
SELECT username
FROM users
WHERE role = 'admin';

-- 05. Módosítsd egy adott eszköz nevét (pl. az id=3 esetén)!
UPDATE devices
SET device_name = 'uj eszkoz nev'
WHERE id = 3;




CREATE DATABASE automated_systems_data;

\c automated_systems_data;

----------------------------------
---- ENUM típusok létrehozása ----
----------------------------------

CREATE TYPE device_type AS ENUM ('sensor', 'controller', 'actuator');
CREATE TYPE device_status AS ENUM ('active', 'inactive', 'maintenance');
CREATE TYPE user_role AS ENUM ('admin', 'user', 'viewer');

----------------------------
---- Táblák létrehozása ----
----------------------------

CREATE TABLE devices (
  id INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  device_name VARCHAR(100) NOT NULL,
  location VARCHAR(100),
  type device_type NOT NULL,
  status device_status NOT NULL DEFAULT 'active',
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE events (
  id INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  device_id INTEGER NOT NULL,
  event_type VARCHAR(50) NOT NULL,
  description TEXT,
  occurred_at TIMESTAMPTZ DEFAULT NOW(),

  CONSTRAINT fk_events_device
    FOREIGN KEY (device_id)
    REFERENCES devices(id)
    ON DELETE CASCADE
);

CREATE TABLE sensor_data (
  id INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  device_id INTEGER NOT NULL,
  data_value NUMERIC(4,1) NOT NULL,
  unit VARCHAR(20),
  recorded_at TIMESTAMPTZ DEFAULT NOW(),

  CONSTRAINT fk_sensor_device
    FOREIGN KEY (device_id)
    REFERENCES devices(id)
    ON DELETE CASCADE
);

CREATE TABLE users (
  id INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  username VARCHAR(50) NOT NULL UNIQUE,
  password_hash VARCHAR(255) NOT NULL,
  role user_role NOT NULL,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-----------------
---- Indexek ----
-----------------
-- PostreSQL automatikusan készít indexet: PRIMARY KEY, UNIQUE, de foreign key mezőkre nem mindig optimális, ezért:

CREATE INDEX idx_events_device_id ON events(device_id);
CREATE INDEX idx_sensor_data_device_id ON sensor_data(device_id);
