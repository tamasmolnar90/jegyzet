////////// ADATBÁZIS LÉTREHOZÁSA //////////

CREATE DATABASE db_name
    ENCODING 'UTF8';

Postgres alapból kezeli az ékezetes karaktereket UTF8-ban.



////////// ADATBÁZIS HASZNÁLATA //////////

pgAdmin4-ben a kívánt adatbázishoz kell csatlakozni, majd azon belül megnyitni a Query Tool-t.

psql esetén:
\c db_name



////////// TÁBLÁK LEKÉRDEZÉSE //////////

Query Tool:
SELECT table_name
FROM information_schema.tables
WHERE table_schema = 'public'
  AND table_type = 'BASE TABLE';

PSQL:
\dt



////////// TÁBLA SZERKEZETÉNEK LEKÉRDEZÉSE //////////

Query Tool:
SELECT
    column_name,
    data_type,
    is_nullable,
    column_default
FROM information_schema.columns
WHERE table_schema = 'public'
  AND table_name = 'table_name'
ORDER BY ordinal_position;

PSQL:
\d table_name
