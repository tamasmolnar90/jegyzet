namespace Teszt {
    public class Szamologep(int szam1, int szam2) {
        public int Szam1 { get; set; } = szam1;
        public int Szam2 { get; set; } = szam2;

        public int Osszead() {
            return Szam1 + Szam2;
        }

        public int Kivonas() {
            return Szam1 - Szam2;
        }

        public int Szorzas() {
            return Szam1 * Szam2;
        }

        public double Osztas() {
            if (Szam2 == 0) {
                //return double.NaN;
                throw new ArgumentException("Nem lehet osztani 0-val");
            }

            //return Convert.ToDouble(Szam1) / Convert.ToDouble(Szam2);
            return (double)Szam1 / Szam2;
        }
    }
}
