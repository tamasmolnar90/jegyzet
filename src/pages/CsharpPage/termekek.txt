namespace _10_Termekek
{
    internal class Termek
    {
        public int Id { get; set; }
        public string Megnevezes { get; set; }
        public string Kategoria { get; set; }
        public double Ar { get; set; }
        public int Keszlet { get; set; }

        public Termek(int id, string megnevezes, string kategoria, double ar, int keszlet)
        {
            Id = id;
            Megnevezes = megnevezes;
            Kategoria = kategoria;
            Ar = ar;
            Keszlet = keszlet;
        }

        public double EuroToHuf(double arfolyam)
        {
            double forint = Ar * arfolyam;
            return Math.Round(forint, 0);
        }

        public double AkciosAr(double ar, double szazalek)
        {
            double akciosAr = ar * (1 - (szazalek / 100));
            return Math.Round(akciosAr, 2);
        }

        public string Formazott()
        {
            return $"id: {Id}, megnevezés: {Megnevezes}, kategória: {Kategoria}, készleten: {Keszlet} darab, termék ára: {Ar} EURO";
        }

        public override string ToString()
        {
            return $"id: {Id}, megnevezés: {Megnevezes}, kategória: {Kategoria}, készleten: {Keszlet} darab, termék ára: {Ar} EURO";
        }
    }
}
