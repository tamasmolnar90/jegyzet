namespace _11_termekek_linq
{
    internal class Termek(int id, string megnevezes, string kategoria, double ar, int keszlet)
    {
        public int Id { get; set; } = id;
        public string Megnevezes { get; set; } = megnevezes;
        public string Kategoria { get; set; } = kategoria;
        public double Ar { get; set; } = ar;
        public int Keszlet { get; set; } = keszlet;

        public double EuroToHuf(double arfolyam)
        {
            double forint = Ar * arfolyam;
            return Math.Round(forint, 0);
        }

        public double AkciosForint(double arfolyam, double szazalek)
        {
            double forint = EuroToHuf(arfolyam);
            double akciosAr = forint * (1 - (szazalek / 100));
            return Math.Round(akciosAr, 0);
        }

        public double AkciosAr(double szazalek)
        {
            double akciosAr = Ar * (1 - (szazalek / 100));
            return Math.Round(akciosAr, 2);
        }

        public string Formazott()
        {
            return $"id: {Id}, megnevezés: {Megnevezes}, kategória: {Kategoria}, készleten: {Keszlet} darab, termék ára: {Ar:F2} EURO";
        }

        public override string ToString()
        {
            return $"id: {Id}, megnevezés: {Megnevezes}, kategória: {Kategoria}, készleten: {Keszlet} darab, termék ára: {Ar:F2} EURO";
        }
    }
}

                .GroupBy(sor => sor.Kategoria)
                .Select(g => (Kategoria: g.Key, Keszlet: g.Sum(sor => sor.Keszlet)))

                })
                .ToList();

            //Hozz létre egy dragatermekek.csv fájlt, amelybe csak az átlag ár feletti termékek kerülnek bele
            var dragaTermekek = termekLista
                .Where(sor => sor.Ar > atlag)
                .Select(sor => $"{sor.Id};{sor.Megnevezes};{sor.Kategoria};{sor.Ar};{sor.Keszlet}")
                .Prepend(header);

            File.WriteAllLines("dragatermekek.csv", dragaTermekek);
        }
    }
