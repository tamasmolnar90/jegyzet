//Konzolra termékek neve és ára (27%-AL NÖVELVE)
//Ha az ár meghaladja a 10.000 Ft-ot a sor végére : "PRÉMIUM"
const termekek = [
  { nev: "Egér", ar: 4000 },
  { nev: "Billentyűzet", ar: 9000 },
  { nev: "Monitor", ar: 45000 },
  { nev: "Egérpad", ar: 2000 },
];

termekek.forEach((termek) => {
  const afasAr = termek.ar * 1.27;

  if (afasAr > 10000) {
    console.log(`${termek.nev} Bruttó ára: ${afasAr} Ft PRÉMIUM`);
  } else {
    console.log(`${termek.nev} Bruttó ára: ${afasAr} Ft`);
  }
});

//Készítsünk szűrést, ami kigyűjti a még el nem végzett feladatokat
//Írassuk ki, hogy hány darab feladat vár még elvégzésre
const feladatok = [
  { cim: "Szoftverfejlesztés házifeladat", kesz: true },
  { cim: "Takarítás", kesz: false },
  { cim: "Projektmunka tervezés", kesz: false },
  { cim: "Bevásárlás ", kesz: true },
];

const befejezetlen = feladatok.filter((feladat) => !feladat.kesz);

console.log(`Hátralévő feladatok listája: ${befejezetlen}`);
console.log(`Még: ${befejezetlen.length} db feladat vár elvégzésre.`);

//map
const uj = [1, 2, 3].map((x) => x * 2);

console.log(uj); // [2, 4, 6]



//filter
const uj = [1, 2, 3, 4].filter((x) => x > 2);

console.log(uj); // [3, 4]

//reduce
const osszeg = [1, 2, 3, 4].reduce((acc, x) => acc + x, 0);

console.log(osszeg); // 10



//find
const talalat = [1, 2, 3, 4].find((x) => x > 2);

console.log(talalat); // 3



//findIndex
const index = [1, 2, 3, 4].findIndex((x) => x > 2);

console.log(index); // 2



//some
const vanParos = [1, 3, 4].some((x) => x % 2 === 0);

console.log(vanParos); // true



//every
const mindParos = [2, 4, 6].every((x) => x % 2 === 0);

console.log(mindParos); // true



//includes
const van = [1, 2, 3].includes(2);

console.log(van); // true



//sort
const szamok = [5, 2, 8];

szamok.sort((a, b) => a - b);

console.log(szamok); // [2, 5, 8]



//slice
const uj = [1, 2, 3, 4, 5].slice(1, 4);

console.log(uj); // [2, 3, 4]



//pop
const tomb = [1, 2, 3];

const utolso = tomb.pop();

console.log(utolso); // 3
