export function add(a, b) {
  return a + b;
}

export function subtract(a, b) {
  return a - b;
}

export function multiply(a, b) {
  return a * b;
}

export function divide(a, b) {
  if (typeof a !== "number" || typeof b !== "number") {
    throw new Error("Both values must be numbers");
  }

  if (b === 0) {
    throw new Error("Cannot divide by zero");
  }

  return a / b;
}

import { describe, test, expect } from "vitest";
import {
  add,
  subtract,
  multiply,
  divide,
} from "../../src/functions/calculator.js";

describe("Calculator", () => {
  // ADD
  test("1 + 2 should equal 3", () => {
    expect(add(1, 2)).toBe(3);
  });

  // SUBTRACT
  test("1 - 2 should equal -1", () => {
    expect(subtract(1, 2)).toBe(-1);
  });

  // MULTIPLY
  test("1 * 2 should equal 2", () => {
    expect(multiply(1, 2)).toBe(2);
  });

  // DIVIDE
  test("return int: 10 / 2 should equal 5", () => {
    expect(divide(10, 2)).toBe(5);
  });

  test("return decimal: 1 / 2 should equal 0.5", () => {
    expect(divide(1, 2)).toBeCloseTo(0.5);
  });

  test("throws error when dividing by zero: 10 / 0", () => {
    expect(() => {
      divide(10, 0);
    }).toThrow("Cannot divide by zero");
  });

  test("throws error when first value is not a number", () => {
    expect(() => {
      divide("10", 2);
    }).toThrow("Both values must be numbers");
  });
});

function isLoggedIn(user) {
  return user !== null;
}

// TRUTHY
test("user is logged in", () => {
  const user = {
    name: "Vision",
  };

  expect(isLoggedIn(user)).toBeTruthy();
});

// FALSY
test("user is not logged in", () => {
  const user = null;

  expect(isLoggedIn(user)).toBeFalsy();
});
